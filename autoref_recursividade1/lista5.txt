5 -> O modelo de função para o tipo lista é da seguinte forma: 
pub fn fn_para_lista(lst: Lista) {
    case lst {
        Vazia -> todo
        NaoVazia(primeiro, resto) -> {
            todo 
            primeiro
            fn_para_lista(resto) 
        }
    }
}
